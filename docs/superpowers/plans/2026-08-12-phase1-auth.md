# Phase 1: Auth Pages (Login · Register · OTP · Forgot Password)

> **For agentic workers:** Use superpowers:executing-plans to implement task-by-task.
> **Prerequisite:** Phase 0 phải hoàn thành trước (middleware + auth API layer).

**Goal:** 4 trang auth hoàn chỉnh, thay thế JSP auth — người dùng có thể đăng nhập/đăng ký qua Next.js.

**Architecture:** Gọi backend `/dangnhap` với AJAX header → JSON response + session cookie → middleware redirect.

**Tech Stack:** Next.js 14 · TypeScript · Tailwind CSS · Lucide React · Java Servlet session

## Global Constraints

- Design: nền gradient `from-vs-navy via-[#1e3a5f] to-vs-blue` (giống HeroSection landing page)
- Card trắng, `rounded-2xl shadow-xl`, max-width `md:w-[420px]`
- Logo V-SPORT trên đầu card
- Error message hiển thị inline (không toast)
- Sau login thành công → redirect theo role (từ `?redirect=` param hoặc role default)
- Tất cả component auth là `'use client'`

```tsx
// vsport_frontend/components/auth/LoginCard.tsx
'use client'

import { useState } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import Link from 'next/link'
import { Mail, Phone, Lock, Eye, EyeOff, AlertCircle } from 'lucide-react'
import { login } from '@/lib/api/auth'

const ROLE_HOME: Record<string, string> = {
  CUSTOMER: '/tim-kiem',
  MANAGER: '/manager/dashboard',
  STAFF: '/staff/dashboard',
  ADMIN: '/admin/dashboard',
}

export default function LoginCard() {
  const router = useRouter()
  const searchParams = useSearchParams()
  const redirectTo = searchParams.get('redirect') ?? ''

  const [method, setMethod] = useState<'account' | 'phone'>('account')
  const [identifier, setIdentifier] = useState('')
  const [password, setPassword] = useState('')
  const [showPassword, setShowPassword] = useState(false)
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!identifier.trim() || !password.trim()) {
      setError('Vui lòng nhập đầy đủ thông tin.')
      return
    }
    setLoading(true)
    setError('')
    try {
      const result = await login(identifier, password, method)
      if (result.success && result.user) {
        const dest = redirectTo || ROLE_HOME[result.user.role] || '/'
        router.push(dest)
      } else {
        setError(result.loi ?? 'Đăng nhập thất bại.')
      }
    } catch {
      setError('Không thể kết nối đến máy chủ.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="w-full md:w-[420px] bg-white rounded-2xl shadow-xl p-8">
      <div className="text-center mb-8">
        <Link href="/owner" className="inline-flex font-bold text-2xl tracking-tight">
          <span className="text-vs-navy">V-</span>
          <span className="text-vs-cyan">SPORT</span>
        </Link>
        <h1 className="mt-4 text-xl font-bold text-vs-navy">Đăng nhập</h1>
      </div>

      {/* Tab switch */}
      <div className="flex rounded-lg border border-slate-200 mb-6 overflow-hidden">
        {(['account', 'phone'] as const).map((m) => (
          <button
            key={m}
            type="button"
            onClick={() => { setMethod(m); setIdentifier(''); setError('') }}
            className={`flex-1 py-2 text-sm font-medium transition-colors ${
              method === m ? 'bg-vs-blue text-white' : 'text-vs-slate hover:bg-slate-50'
            }`}
          >
            {m === 'account' ? 'Email / Tài khoản' : 'Số điện thoại'}
          </button>
        ))}
      </div>

      <form onSubmit={handleSubmit} noValidate className="space-y-4">
        <div className="relative">
          <span className="absolute left-3 top-1/2 -translate-y-1/2 text-vs-slate">
            {method === 'phone' ? <Phone className="w-4 h-4" /> : <Mail className="w-4 h-4" />}
          </span>
          <input
            type={method === 'phone' ? 'tel' : 'text'}
            placeholder={method === 'phone' ? '0901 234 567' : 'Email hoặc tên tài khoản'}
            value={identifier}
            onChange={e => setIdentifier(e.target.value)}
            className="w-full pl-10 pr-4 py-3 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue"
            autoComplete={method === 'phone' ? 'tel' : 'username'}
          />
        </div>

        <div className="relative">
          <Lock className="absolute left-3 top-1/2 -translate-y-1/2 text-vs-slate w-4 h-4" />
          <input
            type={showPassword ? 'text' : 'password'}
            placeholder="Mật khẩu"
            value={password}
            onChange={e => setPassword(e.target.value)}
            className="w-full pl-10 pr-10 py-3 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue"
            autoComplete="current-password"
          />
          <button
            type="button"
            onClick={() => setShowPassword(v => !v)}
            className="absolute right-3 top-1/2 -translate-y-1/2 text-vs-slate"
            tabIndex={-1}
          >
            {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
          </button>
        </div>

        {error && (
          <div className="flex items-center gap-2 bg-red-50 border border-red-200 text-red-600 text-sm px-3 py-2.5 rounded-lg">
            <AlertCircle className="w-4 h-4 shrink-0" />
            {error}
          </div>
        )}

        <button
          type="submit"
          disabled={loading}
          className="w-full bg-vs-blue hover:bg-vs-navy disabled:opacity-60 text-white font-semibold py-3 rounded-xl transition-colors"
        >
          {loading ? 'Đang đăng nhập…' : 'Đăng nhập'}
        </button>
      </form>

      <div className="mt-4 text-center text-sm text-vs-slate">
        <Link href="/quen-mat-khau" className="hover:text-vs-blue transition-colors">
          Quên mật khẩu?
        </Link>
      </div>

      <div className="mt-6 text-center text-sm text-vs-slate">
        Chưa có tài khoản?{' '}
        <Link href="/dang-ky" className="text-vs-blue font-medium hover:underline">
          Đăng ký ngay
        </Link>
      </div>
    </div>
  )
}
```

- [ ] **Step 1: Viết RegisterCard**

```tsx
// vsport_frontend/components/auth/RegisterCard.tsx
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import { User, Mail, Phone, Lock, Eye, EyeOff, AlertCircle } from 'lucide-react'

const BASE = process.env.NEXT_PUBLIC_BACKEND_URL ?? 'http://localhost:8080/Backend_java'
const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
const VN_PHONE_RE = /^(0[35789])\d{8}$/

export default function RegisterCard() {
  const router = useRouter()
  const [form, setForm] = useState({ hoTen: '', email: '', soDienThoai: '', matKhau: '', xacNhanMatKhau: '' })
  const [showPwd, setShowPwd] = useState(false)
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const set = (field: keyof typeof form) => (e: React.ChangeEvent<HTMLInputElement>) =>
    setForm(f => ({ ...f, [field]: e.target.value }))

  const validate = (): string => {
    if (!form.hoTen.trim()) return 'Vui lòng nhập họ tên.'
    if (!EMAIL_RE.test(form.email)) return 'Email không hợp lệ.'
    if (!VN_PHONE_RE.test(form.soDienThoai)) return 'Số điện thoại không hợp lệ. Nhập đúng định dạng 0xxx xxxxxxx.'
    if (form.matKhau.length < 6) return 'Mật khẩu tối thiểu 6 ký tự.'
    if (form.matKhau !== form.xacNhanMatKhau) return 'Mật khẩu xác nhận không khớp.'
    return ''
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    const err = validate()
    if (err) { setError(err); return }
    setLoading(true); setError('')
    try {
      const res = await fetch(`${BASE}/dangky`, {
        method: 'POST',
        credentials: 'include',
        headers: {
          'Content-Type': 'application/x-www-form-urlencoded',
          'X-Requested-With': 'XMLHttpRequest',
        },
        body: new URLSearchParams(form).toString(),
      })
      const data = await res.json()
      if (data.success) {
        router.push(`/xac-thuc-otp?email=${encodeURIComponent(form.email)}`)
      } else {
        setError(data.loi ?? 'Đăng ký thất bại.')
      }
    } catch {
      setError('Không thể kết nối đến máy chủ.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="w-full md:w-[460px] bg-white rounded-2xl shadow-xl p-8">
      <div className="text-center mb-6">
        <Link href="/" className="inline-flex font-bold text-2xl tracking-tight">
          <span className="text-vs-navy">V-</span><span className="text-vs-cyan">SPORT</span>
        </Link>
        <h1 className="mt-3 text-xl font-bold text-vs-navy">Tạo tài khoản</h1>
      </div>

      <form onSubmit={handleSubmit} noValidate className="space-y-4">
        {[
          { field: 'hoTen', placeholder: 'Họ và tên', icon: <User className="w-4 h-4" />, type: 'text' },
          { field: 'email', placeholder: 'Email', icon: <Mail className="w-4 h-4" />, type: 'email' },
          { field: 'soDienThoai', placeholder: '0901 234 567', icon: <Phone className="w-4 h-4" />, type: 'tel' },
        ].map(({ field, placeholder, icon, type }) => (
          <div key={field} className="relative">
            <span className="absolute left-3 top-1/2 -translate-y-1/2 text-vs-slate">{icon}</span>
            <input
              type={type}
              placeholder={placeholder}
              value={form[field as keyof typeof form]}
              onChange={set(field as keyof typeof form)}
              className="w-full pl-10 pr-4 py-3 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue"
            />
          </div>
        ))}

        {['matKhau', 'xacNhanMatKhau'].map((field, i) => (
          <div key={field} className="relative">
            <Lock className="absolute left-3 top-1/2 -translate-y-1/2 text-vs-slate w-4 h-4" />
            <input
              type={showPwd ? 'text' : 'password'}
              placeholder={i === 0 ? 'Mật khẩu' : 'Xác nhận mật khẩu'}
              value={form[field as keyof typeof form]}
              onChange={set(field as keyof typeof form)}
              className="w-full pl-10 pr-10 py-3 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue"
            />
            {i === 0 && (
              <button type="button" onClick={() => setShowPwd(v => !v)} className="absolute right-3 top-1/2 -translate-y-1/2 text-vs-slate" tabIndex={-1}>
                {showPwd ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
              </button>
            )}
          </div>
        ))}

        {error && (
          <div className="flex items-center gap-2 bg-red-50 border border-red-200 text-red-600 text-sm px-3 py-2.5 rounded-lg">
            <AlertCircle className="w-4 h-4 shrink-0" />{error}
          </div>
        )}

        <button type="submit" disabled={loading}
          className="w-full bg-vs-blue hover:bg-vs-navy disabled:opacity-60 text-white font-semibold py-3 rounded-xl transition-colors">
          {loading ? 'Đang xử lý…' : 'Đăng ký'}
        </button>
      </form>

      <p className="mt-6 text-center text-sm text-vs-slate">
        Đã có tài khoản?{' '}
        <Link href="/dang-nhap" className="text-vs-blue font-medium hover:underline">Đăng nhập</Link>
      </p>
    </div>
  )
}
```

### Task 3: OTP Verification Page (`/xac-thuc-otp`)

**Files:**
- Create: `vsport_frontend/components/auth/OtpCard.tsx`
- Create: `vsport_frontend/app/(public)/xac-thuc-otp/page.tsx`

**Logic:** Nhận `?email=` từ URL. 6 ô input riêng biệt (giống Step2OTP trong owner-register).
Gọi `POST /nhapma` với `email` + `ma` (OTP code).

- [ ] **Step 1: Viết OtpCard**

```tsx
// vsport_frontend/components/auth/OtpCard.tsx
'use client'

import { useState, useRef, useEffect } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import Link from 'next/link'
import { AlertCircle } from 'lucide-react'

const BASE = process.env.NEXT_PUBLIC_BACKEND_URL ?? 'http://localhost:8080/Backend_java'

export default function OtpCard() {
  const router = useRouter()
  const searchParams = useSearchParams()
  const email = searchParams.get('email') ?? ''

  const [digits, setDigits] = useState(['', '', '', '', '', ''])
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)
  const [countdown, setCountdown] = useState(300)
  const refs = useRef<(HTMLInputElement | null)[]>([])

  useEffect(() => {
    if (countdown <= 0) return
    const t = setTimeout(() => setCountdown(c => c - 1), 1000)
    return () => clearTimeout(t)
  }, [countdown])

  const otp = digits.join('')

  const handleChange = (i: number, val: string) => {
    if (!/^\d?$/.test(val)) return
    const next = [...digits]
    next[i] = val
    setDigits(next)
    if (val && i < 5) refs.current[i + 1]?.focus()
    if (next.every(d => d !== '')) submitOtp(next.join(''))
  }

  const handleKeyDown = (i: number, e: React.KeyboardEvent) => {
    if (e.key === 'Backspace' && !digits[i] && i > 0) refs.current[i - 1]?.focus()
  }

  const submitOtp = async (code: string) => {
    setLoading(true); setError('')
    try {
      const res = await fetch(`${BASE}/nhapma`, {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'X-Requested-With': 'XMLHttpRequest' },
        body: new URLSearchParams({ email, ma: code }).toString(),
      })
      const data = await res.json()
      if (data.success) {
        router.push('/dang-nhap')
      } else {
        setError(data.loi ?? 'Mã OTP không đúng.')
        setDigits(['', '', '', '', '', ''])
        refs.current[0]?.focus()
      }
    } catch {
      setError('Không thể kết nối đến máy chủ.')
    } finally {
      setLoading(false)
    }
  }

  const resend = async () => {
    if (countdown > 0) return
    await fetch(`${BASE}/resend-otp`, { method: 'POST', credentials: 'include', headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'X-Requested-With': 'XMLHttpRequest' }, body: new URLSearchParams({ email }).toString() })
    setCountdown(300)
  }

  const mm = String(Math.floor(countdown / 60)).padStart(2, '0')
  const ss = String(countdown % 60).padStart(2, '0')

  return (
    <div className="w-full md:w-[400px] bg-white rounded-2xl shadow-xl p-8 text-center">
      <Link href="/" className="inline-flex font-bold text-2xl tracking-tight mb-6">
        <span className="text-vs-navy">V-</span><span className="text-vs-cyan">SPORT</span>
      </Link>
      <h1 className="text-xl font-bold text-vs-navy mb-2">Xác thực email</h1>
      <p className="text-vs-slate text-sm mb-6">
        Nhập mã 6 chữ số đã gửi đến<br />
        <span className="font-medium text-vs-navy">{email}</span>
      </p>

      <div className="flex justify-center gap-2 mb-6">
        {digits.map((d, i) => (
          <input
            key={i}
            ref={el => { refs.current[i] = el }}
            value={d}
            onChange={e => handleChange(i, e.target.value)}
            onKeyDown={e => handleKeyDown(i, e)}
            maxLength={1}
            inputMode="numeric"
            className="w-11 h-12 text-center text-lg font-bold border-2 border-slate-200 rounded-xl focus:border-vs-blue focus:outline-none transition-colors"
          />
        ))}
      </div>

      {error && (
        <div className="flex items-center gap-2 bg-red-50 border border-red-200 text-red-600 text-sm px-3 py-2.5 rounded-lg mb-4 text-left">
          <AlertCircle className="w-4 h-4 shrink-0" />{error}
        </div>
      )}

      <p className="text-sm text-vs-slate mb-2">
        Mã hết hạn sau <span className="font-mono font-semibold text-vs-navy">{mm}:{ss}</span>
      </p>

      <button onClick={resend} disabled={countdown > 0}
        className="text-sm text-vs-blue hover:underline disabled:text-vs-slate disabled:cursor-not-allowed">
        Gửi lại mã
      </button>
    </div>
  )
}
```

### Task 4: Forgot Password Page (`/quen-mat-khau`)

**Files:**
- Create: `vsport_frontend/components/auth/ForgotPasswordCard.tsx`
- Create: `vsport_frontend/app/(public)/quen-mat-khau/page.tsx`

**Logic:** 2 bước — (1) nhập email, (2) nhập OTP + mật khẩu mới.
Gọi `POST /quenmatkhau` với email; step 2 gọi `POST /quenmatkhau` với ma + matKhauMoi.

- [ ] **Step 1: Viết ForgotPasswordCard**

```tsx
// vsport_frontend/components/auth/ForgotPasswordCard.tsx
'use client'

import { useState } from 'react'
import Link from 'next/link'
import { useRouter } from 'next/navigation'
import { Mail, Lock, AlertCircle, CheckCircle } from 'lucide-react'

const BASE = process.env.NEXT_PUBLIC_BACKEND_URL ?? 'http://localhost:8080/Backend_java'

export default function ForgotPasswordCard() {
  const router = useRouter()
  const [step, setStep] = useState<1 | 2>(1)
  const [email, setEmail] = useState('')
  const [otp, setOtp] = useState('')
  const [newPassword, setNewPassword] = useState('')
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const sendOtp = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!email.trim()) { setError('Vui lòng nhập email.'); return }
    setLoading(true); setError('')
    try {
      const res = await fetch(`${BASE}/quenmatkhau`, {
        method: 'POST', credentials: 'include',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'X-Requested-With': 'XMLHttpRequest' },
        body: new URLSearchParams({ email, action: 'send' }).toString(),
      })
      const data = await res.json()
      if (data.success) setStep(2)
      else setError(data.loi ?? 'Gửi OTP thất bại.')
    } catch {
      setError('Không thể kết nối.')
    } finally { setLoading(false) }
  }

  const resetPassword = async (e: React.FormEvent) => {
    e.preventDefault()
    if (otp.length < 6) { setError('Nhập đủ 6 chữ số OTP.'); return }
    if (newPassword.length < 6) { setError('Mật khẩu tối thiểu 6 ký tự.'); return }
    setLoading(true); setError('')
    try {
      const res = await fetch(`${BASE}/quenmatkhau`, {
        method: 'POST', credentials: 'include',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'X-Requested-With': 'XMLHttpRequest' },
        body: new URLSearchParams({ email, ma: otp, matKhauMoi: newPassword, action: 'reset' }).toString(),
      })
      const data = await res.json()
      if (data.success) router.push('/dang-nhap')
      else setError(data.loi ?? 'Đặt lại mật khẩu thất bại.')
    } catch {
      setError('Không thể kết nối.')
    } finally { setLoading(false) }
  }

  return (
    <div className="w-full md:w-[400px] bg-white rounded-2xl shadow-xl p-8">
      <div className="text-center mb-6">
        <Link href="/" className="inline-flex font-bold text-2xl tracking-tight">
          <span className="text-vs-navy">V-</span><span className="text-vs-cyan">SPORT</span>
        </Link>
        <h1 className="mt-3 text-xl font-bold text-vs-navy">Quên mật khẩu</h1>
      </div>

      {step === 1 ? (
        <form onSubmit={sendOtp} className="space-y-4">
          <div className="relative">
            <Mail className="absolute left-3 top-1/2 -translate-y-1/2 text-vs-slate w-4 h-4" />
            <input type="email" placeholder="Email đã đăng ký" value={email}
              onChange={e => setEmail(e.target.value)}
              className="w-full pl-10 pr-4 py-3 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue" />
          </div>
          {error && <div className="flex items-center gap-2 bg-red-50 border border-red-200 text-red-600 text-sm px-3 py-2.5 rounded-lg"><AlertCircle className="w-4 h-4 shrink-0" />{error}</div>}
          <button type="submit" disabled={loading}
            className="w-full bg-vs-blue hover:bg-vs-navy disabled:opacity-60 text-white font-semibold py-3 rounded-xl transition-colors">
            {loading ? 'Đang gửi…' : 'Gửi mã xác nhận'}
          </button>
        </form>
      ) : (
        <form onSubmit={resetPassword} className="space-y-4">
          <p className="text-sm text-vs-slate text-center">Nhập mã OTP đã gửi đến <strong>{email}</strong></p>
          <input type="text" placeholder="Mã OTP (6 chữ số)" value={otp}
            onChange={e => setOtp(e.target.value.replace(/\D/g, '').slice(0, 6))}
            className="w-full px-4 py-3 border border-slate-200 rounded-xl text-sm text-center tracking-widest font-mono focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue" />
          <div className="relative">
            <Lock className="absolute left-3 top-1/2 -translate-y-1/2 text-vs-slate w-4 h-4" />
            <input type="password" placeholder="Mật khẩu mới" value={newPassword}
              onChange={e => setNewPassword(e.target.value)}
              className="w-full pl-10 pr-4 py-3 border border-slate-200 rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-vs-blue/30 focus:border-vs-blue" />
          </div>
          {error && <div className="flex items-center gap-2 bg-red-50 border border-red-200 text-red-600 text-sm px-3 py-2.5 rounded-lg"><AlertCircle className="w-4 h-4 shrink-0" />{error}</div>}
          <button type="submit" disabled={loading}
            className="w-full bg-vs-blue hover:bg-vs-navy disabled:opacity-60 text-white font-semibold py-3 rounded-xl transition-colors">
            {loading ? 'Đang xử lý…' : 'Đặt lại mật khẩu'}
          </button>
        </form>
      )}

      <p className="mt-6 text-center text-sm">
        <Link href="/dang-nhap" className="text-vs-blue hover:underline">← Quay lại đăng nhập</Link>
      </p>
    </div>
  )
}
```
